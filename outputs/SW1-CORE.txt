enable
configure terminal
hostname SW1-CORE

! Crear VLANs
vlan 10
 name ADMINISTRACION
vlan 20
 name VENTAS
vlan 30
 name SERVIDORES
vlan 40
 name INVITADOS

! Puerto trunk hacia R1
interface g0/1
 description TRUNK-HACIA-R1-CORE
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40
 no shutdown
exit

! Puertos access
interface f0/1
 description PC-ADMINISTRACION
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
exit

interface f0/2
 description PC-VENTAS
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
exit

interface f0/3
 description SERVIDOR
 switchport mode access
 switchport access vlan 30
 spanning-tree portfast
exit

interface f0/4
 description PC-INVITADOS
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
exit

end
write memory
