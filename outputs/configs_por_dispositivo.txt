=== NETFORGE: CONFIGURACIÓN POR DISPOSITIVO ===

=== SW1-CORE ===
enable
configure terminal
hostname SW1-CORE

! VLANs
vlan 10
 name ADMINISTRACION
vlan 20
 name VENTAS
vlan 30
 name SERVIDORES
vlan 40
 name INVITADOS

end
write memory

=== R1-CORE ===
enable
configure terminal
hostname R1-CORE

! DHCP Pools
ip dhcp pool ADMINISTRACION
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8

ip dhcp pool VENTAS
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 8.8.8.8

ip dhcp pool SERVIDORES
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 8.8.8.8

ip dhcp pool INVITADOS
 network 192.168.40.0 255.255.255.0
 default-router 192.168.40.1
 dns-server 8.8.8.8

! OSPF
router ospf 1
 router-id 1.1.1.1
 network 192.168.0.0 0.0.255.255 area 0
 network 10.0.0.0 0.0.0.255 area 0

! NAT
access-list 1 permit 192.168.0.0 0.0.255.255
interface g0/0
 ip nat outside
exit
interface g0/1
 ip nat inside
exit
ip nat inside source list 1 interface g0/0 overload

end
write memory

=== R2-REMOTA ===
enable
configure terminal
hostname R2-REMOTA

! OSPF remoto
router ospf 1
 router-id 2.2.2.2
 network 192.168.50.0 0.0.0.255 area 0
 network 10.0.0.0 0.0.0.255 area 0

end
write memory

=== R-ISP ===
enable
configure terminal
hostname R-ISP

! Config básica ISP simulada
interface g0/0
 description Enlace hacia R1-CORE
 no shutdown
exit
interface g0/1
 description Enlace hacia R2-REMOTA o Internet simulado
 no shutdown
exit

end
write memory
